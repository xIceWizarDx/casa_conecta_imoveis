USE `gabr2831_casa_conecta_imoveis`;

SET NAMES utf8;
SET FOREIGN_KEY_CHECKS = 0;

-- Tabela: users
CREATE TABLE IF NOT EXISTS `users` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `name` VARCHAR(255) NOT NULL,
  `email` VARCHAR(191) NOT NULL,
  `email_verified_at` DATETIME NULL,
  `password` VARCHAR(255) NOT NULL,
  `remember_token` VARCHAR(100) NULL,
  `created_at` DATETIME NULL,
  `updated_at` DATETIME NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `users_email_unique` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

-- Tabela: password_reset_tokens
CREATE TABLE IF NOT EXISTS `password_reset_tokens` (
  `email` VARCHAR(191) NOT NULL,
  `token` VARCHAR(255) NOT NULL,
  `created_at` DATETIME NULL,
  PRIMARY KEY (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

-- Tabela: sessions (Laravel Session Driver: database)
CREATE TABLE IF NOT EXISTS `sessions` (
  `id` VARCHAR(191) NOT NULL,
  `user_id` BIGINT UNSIGNED NULL,
  `ip_address` VARCHAR(45) NULL,
  `user_agent` TEXT NULL,
  `payload` LONGTEXT NOT NULL,
  `last_activity` INT NOT NULL,
  PRIMARY KEY (`id`),
  KEY `sessions_user_id_index` (`user_id`),
  KEY `sessions_last_activity_index` (`last_activity`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

-- Tabela: cache (Laravel Cache Store: database)
CREATE TABLE IF NOT EXISTS `cache` (
  `key` VARCHAR(191) NOT NULL,
  `value` MEDIUMTEXT NOT NULL,
  `expiration` INT NOT NULL,
  PRIMARY KEY (`key`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

-- Tabela: cache_locks (Laravel Cache Locks)
CREATE TABLE IF NOT EXISTS `cache_locks` (
  `key` VARCHAR(191) NOT NULL,
  `owner` VARCHAR(191) NOT NULL,
  `expiration` INT NOT NULL,
  PRIMARY KEY (`key`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

-- Tabela: jobs (Laravel Queue: database)
CREATE TABLE IF NOT EXISTS `jobs` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `queue` VARCHAR(191) NOT NULL,
  `payload` LONGTEXT NOT NULL,
  `attempts` TINYINT UNSIGNED NOT NULL,
  `reserved_at` INT UNSIGNED NULL,
  `available_at` INT UNSIGNED NOT NULL,
  `created_at` INT UNSIGNED NOT NULL,
  PRIMARY KEY (`id`),
  KEY `jobs_queue_index` (`queue`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

-- Tabela: job_batches (Laravel Batches)
CREATE TABLE IF NOT EXISTS `job_batches` (
  `id` VARCHAR(191) NOT NULL,
  `name` VARCHAR(255) NOT NULL,
  `total_jobs` INT NOT NULL,
  `pending_jobs` INT NOT NULL,
  `failed_jobs` INT NOT NULL,
  `failed_job_ids` LONGTEXT NOT NULL,
  `options` MEDIUMTEXT NULL,
  `cancelled_at` INT NULL,
  `created_at` INT NOT NULL,
  `finished_at` INT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

CREATE TABLE IF NOT EXISTS `settings` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `key` VARCHAR(191) NOT NULL UNIQUE,
  `value` LONGTEXT NULL,
  `created_at` DATETIME NULL,
  `updated_at` DATETIME NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

CREATE TABLE IF NOT EXISTS `images` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `disk` VARCHAR(50) NOT NULL DEFAULT 'public',
  `path` VARCHAR(255) NOT NULL,
  `filename` VARCHAR(191) NOT NULL,
  `original_name` VARCHAR(255) NOT NULL,
  `size` BIGINT UNSIGNED NOT NULL,
  `mime_type` VARCHAR(100) NULL,
  `width` INT UNSIGNED NULL,
  `height` INT UNSIGNED NULL,
  `created_at` DATETIME NULL,
  `updated_at` DATETIME NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `images_filename_unique` (`filename`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

CREATE TABLE IF NOT EXISTS `videos` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `disk` VARCHAR(50) NOT NULL,
  `path` VARCHAR(255) NOT NULL,
  `filename` VARCHAR(191) NOT NULL,
  `original_name` VARCHAR(255) NULL,
  `size` BIGINT UNSIGNED NULL,
  `mime_type` VARCHAR(100) NULL,
  `width` INT UNSIGNED NULL,
  `height` INT UNSIGNED NULL,
  `duration` INT UNSIGNED NULL,
  `created_at` DATETIME NULL,
  `updated_at` DATETIME NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

CREATE TABLE IF NOT EXISTS `hero_slides` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `image_id` BIGINT UNSIGNED NULL,
  `video_id` BIGINT UNSIGNED NULL,
  `title` VARCHAR(255) NOT NULL,
  `subtitle` VARCHAR(255) NULL,
  `price` VARCHAR(100) NOT NULL,
  `bedrooms` INT UNSIGNED NULL,
  `bathrooms` INT UNSIGNED NULL,
  `area` VARCHAR(50) NULL,
  `neighborhood` VARCHAR(100) NULL,
  `is_new` TINYINT(1) NOT NULL DEFAULT 0,
  `is_published` TINYINT(1) NOT NULL DEFAULT 0,
  `position` INT UNSIGNED NOT NULL DEFAULT 0,
  `created_at` DATETIME NULL,
  `updated_at` DATETIME NULL,
  PRIMARY KEY (`id`),
  KEY `hero_slides_image_id_index` (`image_id`),
  KEY `hero_slides_video_id_index` (`video_id`),
  KEY `hero_slides_published_position_index` (`is_published`, `position`),
  CONSTRAINT `hero_slides_image_id_foreign`
    FOREIGN KEY (`image_id`) REFERENCES `images` (`id`)
    ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `hero_slides_video_id_foreign`
    FOREIGN KEY (`video_id`) REFERENCES `videos` (`id`)
    ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

CREATE TABLE IF NOT EXISTS `featured_properties` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `image_id` BIGINT UNSIGNED NOT NULL,
  `title` VARCHAR(255) NOT NULL,
  `neighborhood` VARCHAR(100) NULL,
  `price` VARCHAR(100) NOT NULL,
  `bedrooms` INT UNSIGNED NULL,
  `bathrooms` INT UNSIGNED NULL,
  `area` VARCHAR(50) NULL,
  `built_area` VARCHAR(50) NULL,
  `type` VARCHAR(30) NULL,
  `description` TEXT NULL,
  `features` JSON NULL,
  `price_range` VARCHAR(50) NULL,
  `is_new` TINYINT(1) NOT NULL DEFAULT 0,
  `is_published` TINYINT(1) NOT NULL DEFAULT 0,
  `position` INT UNSIGNED NOT NULL DEFAULT 0,
  `created_at` DATETIME NULL,
  `updated_at` DATETIME NULL,
  PRIMARY KEY (`id`),
  KEY `featured_properties_image_id_index` (`image_id`),
  KEY `featured_properties_published_position_index` (`is_published`, `position`),
  CONSTRAINT `featured_properties_image_id_foreign`
    FOREIGN KEY (`image_id`) REFERENCES `images` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

CREATE TABLE IF NOT EXISTS `featured_property_images` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `featured_property_id` BIGINT UNSIGNED NOT NULL,
  `image_id` BIGINT UNSIGNED NOT NULL,
  `position` INT UNSIGNED NOT NULL DEFAULT 1,
  `created_at` DATETIME NULL,
  `updated_at` DATETIME NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `featured_property_images_unique` (`featured_property_id`, `image_id`),
  KEY `featured_property_images_position_index` (`featured_property_id`, `position`),
  CONSTRAINT `featured_property_images_property_foreign`
    FOREIGN KEY (`featured_property_id`) REFERENCES `featured_properties` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `featured_property_images_image_foreign`
    FOREIGN KEY (`image_id`) REFERENCES `images` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

CREATE TABLE IF NOT EXISTS `featured_property_videos` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `featured_property_id` BIGINT UNSIGNED NOT NULL,
  `video_id` BIGINT UNSIGNED NOT NULL,
  `position` INT UNSIGNED NOT NULL DEFAULT 1,
  `created_at` DATETIME NULL,
  `updated_at` DATETIME NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `featured_property_videos_unique` (`featured_property_id`, `video_id`),
  CONSTRAINT `featured_property_videos_property_foreign`
    FOREIGN KEY (`featured_property_id`) REFERENCES `featured_properties` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `featured_property_videos_video_foreign`
    FOREIGN KEY (`video_id`) REFERENCES `videos` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8 COLLATE=utf8_unicode_ci;

SET FOREIGN_KEY_CHECKS = 1;
